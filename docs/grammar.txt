imp-and-expr      ::= or-expr, { or-expr } ;
or-expr           ::= exp-and-expr, { or-op, exp-and-expr } ;
exp-and-expr      ::= positional-expr, { and-op, positional-expr } ;
positional-expr   ::= not-expr, { positional-op, not-expr }; 

not-expr          ::= [ not-op ], anchored-expr ;
anchored-expr     ::= '^', exact-expr
                    | exact-expr, '$'
                    | exact-expr ;
exact-expr        ::= [ exact-op ], primary-expr ;

primary-expr      ::= literal | phrase | nested-expr ;
nested-expr       ::= '(', imp-and-expr ')' ;
phrase            ::= '"', phrase-exact-expr, { phras-exact-expr }, '"' ;
phrase-exact-expr ::= [ exact-op ], literal ;

literal           ::= term | pattern ;
pattern           ::= { alphabet | wildcard }, wildcard, { alphabet | wildcard } ;
term              ::= alphabet, { alphabet } ;
wildcard          ::= '*' | '?' ;
number            ::= digit, { digit } ;

positional-op     ::= precedence-op | near-op ;
precedence-op     ::= '<<' ;
near-op           ::= '<', near-range, '>' ;
near-range        ::= number, ':', number
                    | number, ':'
                    | ':', number
                    | number ;

and-op            ::= '&' ;
or-op             ::= '|' ;
not-op            ::= '~' ;
exact-op          ::= '=' ;
